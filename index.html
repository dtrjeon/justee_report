GOOGLE SHEETS

Apps Script 웹앱 URL을 입력하면 구글 시트에 데이터가 저장됩니다.

설정 방법 (최초 1회)
1. 📊 공개보고서 구글 시트 열기
2. 확장 프로그램 → Apps Script
3. 아래 코드 복사 → 저장 → 배포 → 새 배포
4. 웹앱 · 모든 사용자 → 배포 → URL 입력
📋 Apps Script 코드 보기 
const SHEET_ID='1uV9myI09dSn_0fhFk9LCXvDHvQuDOU_lEkwegs_Z75c';
const SHEET_NAME='시트1';
function doPost(e){try{const d=JSON.parse(e.postData.contents);const s=SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);s.appendRow([new Date().toLocaleString('ko-KR'),d.jo,d.leader,d.trainer,d.trainee1,d.contact,d.gospel,d.target,d.decision,d.reject,d.confirm,d.believer,d.location,d.name,d.age,d.marriage,d.gender,d.religion,d.job,d.heaven,d.reason,d.yeopjeop,d.diag_result,d.testimony6,d.testimony7,d.leader_part,d.leader_action,d.trainer_part,d.trainer_action,d.trainee_part,d.trainee_action,d.result,d.immed_nurture,d.followup,d.q10,d.q11,d.q12,d.q13,d.week,d.date,d.doc_number,d.evang_type,d.evang_count]);return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);}catch(e){return ContentService.createTextOutput(JSON.stringify({status:'error',message:e.toString()})).setMimeType(ContentService.MimeType.JSON);}}
function doGet(e){try{const s=SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME);const rows=s.getDataRange().getValues();const h=rows[0];const rec=rows.slice(1).map((r,i)=>{const o={rowIndex:i+2};h.forEach((k,j)=>o[k]=r[j]);return o;}).reverse();return ContentService.createTextOutput(JSON.stringify({status:'ok',records:rec})).setMimeType(ContentService.MimeType.JSON);}catch(e){return ContentService.createTextOutput(JSON.stringify({status:'error',message:e.toString()})).setMimeType(ContentService.MimeType.JSON);}}
